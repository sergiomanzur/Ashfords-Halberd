cmake_minimum_required(VERSION 3.25)
project(AshfordsHalberd VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT WIN32)
    message(FATAL_ERROR "This increment targets Windows; the portable game core is a later increment.")
endif()
find_package(SDL2 CONFIG REQUIRED)
add_library(ashford_core src/core.cpp)
target_include_directories(ashford_core PUBLIC src)
target_link_libraries(ashford_core PUBLIC bcrypt)
add_executable(AshfordsHalberd WIN32 src/main.cpp src/text.cpp src/viewer.cpp)
target_link_libraries(AshfordsHalberd PRIVATE ashford_core ashford_assets SDL2::SDL2-static comdlg32 shell32 ole32 gdi32)
target_compile_definitions(AshfordsHalberd PRIVATE SDL_MAIN_HANDLED UNICODE _UNICODE)
if(MINGW)
    target_link_options(AshfordsHalberd PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
foreach(target ashford_core AshfordsHalberd)
    target_compile_options(${target} PRIVATE -Wall -Wextra)
endforeach()
enable_testing()
add_library(ashford_assets src/assets.cpp src/scene.cpp src/render.cpp src/camera.cpp src/camera_render.cpp src/motion.cpp src/actor.cpp src/actor_render.cpp)
target_include_directories(ashford_assets PUBLIC src)
target_compile_options(ashford_assets PRIVATE -Wall -Wextra)
add_executable(asset-tests tests/asset_tests.cpp)
target_link_libraries(asset-tests PRIVATE ashford_assets)
add_test(NAME asset-boundaries COMMAND asset-tests)
add_executable(scene-tests tests/scene_tests.cpp)
target_link_libraries(scene-tests PRIVATE ashford_assets)
add_test(NAME scene-boundaries COMMAND scene-tests)
add_executable(render-tests tests/render_tests.cpp)
target_link_libraries(render-tests PRIVATE ashford_assets)
add_test(NAME raster-boundaries COMMAND render-tests)
add_executable(camera-tests tests/camera_tests.cpp)
target_link_libraries(camera-tests PRIVATE ashford_assets)
add_test(NAME camera-boundaries COMMAND camera-tests)
add_executable(motion-tests tests/motion_tests.cpp)
target_link_libraries(motion-tests PRIVATE ashford_assets)
add_test(NAME motion-boundaries COMMAND motion-tests)
add_executable(actor-tests tests/actor_tests.cpp)
target_link_libraries(actor-tests PRIVATE ashford_assets)
add_test(NAME actor-boundaries COMMAND actor-tests)
add_executable(AshfordPlayerInspect tools/player_inspect.cpp)
target_link_libraries(AshfordPlayerInspect PRIVATE ashford_assets)
if(MINGW)
    target_link_options(AshfordPlayerInspect PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
add_executable(AshfordRoomInspect tools/room_inspect.cpp)
target_link_libraries(AshfordRoomInspect PRIVATE ashford_assets)
if(MINGW)
    target_link_options(AshfordRoomInspect PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
add_executable(AshfordScenePreview tools/scene_preview.cpp)
target_link_libraries(AshfordScenePreview PRIVATE ashford_assets)
if(MINGW)
    target_link_options(AshfordScenePreview PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
add_executable(core-tests tests/core_tests.cpp)
target_link_libraries(core-tests PRIVATE ashford_core)
add_test(NAME core-boundaries COMMAND core-tests "${CMAKE_SOURCE_DIR}/local/core-tests" "${CMAKE_SOURCE_DIR}/local/test-fixtures")
add_test(NAME native-menu-smoke COMMAND AshfordsHalberd --smoke-test --data-dir "${CMAKE_SOURCE_DIR}/local/smoke-user")
set_tests_properties(native-menu-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME native-viewer-smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/viewer_runtime_tests.py" "$<TARGET_FILE:AshfordsHalberd>" "${CMAKE_SOURCE_DIR}/local/viewer-tests")
add_test(NAME packaged-runtime-boundaries COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/runtime_tests.py" "$<TARGET_FILE:AshfordsHalberd>" "${CMAKE_SOURCE_DIR}/local/runtime-tests")
install(TARGETS AshfordsHalberd RUNTIME DESTINATION .)
install(FILES docs/windows-package-readme.txt DESTINATION . RENAME README.txt)
install(FILES external/SDL/LICENSE.txt DESTINATION licenses RENAME SDL-LICENSE.txt)
install(FILES upstream/recvx-decomp/LICENSE DESTINATION licenses RENAME recvx-decomp-LICENSE.txt)
