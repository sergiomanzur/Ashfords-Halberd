cmake_minimum_required(VERSION 3.25)
project(AshfordToolchainSmoke LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
enable_testing()
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL2_DISABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../external/SDL ${CMAKE_BINARY_DIR}/sdl)
add_executable(toolchain-smoke main.c)
target_link_libraries(toolchain-smoke PRIVATE SDL2::SDL2-static)
add_test(NAME native-platform-tools COMMAND toolchain-smoke)
